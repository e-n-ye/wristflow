cmake_minimum_required(VERSION 3.20)
project(wristflow_ui_checks C)
enable_testing()
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(UI "${REPO}/ui/xml")
set(LVGL "${REPO}/vendor/SiFli-SDK/external/lvgl_v9")
set(DEMO "${REPO}/apps/ui_demo/src")
set(RUNTIME "${REPO}/ui/runtime")
set(CORE "${REPO}/core")
option(WRISTFLOW_BUILD_SIMULATOR "Build the official LVGL Win32 driver with the UI demo" OFF)
if(WRISTFLOW_BUILD_SIMULATOR AND NOT WIN32)
    message(FATAL_ERROR "This simulator entry currently requires Windows")
endif()

add_executable(clock_format test_clock_format.c "${UI}/clock_format.c")
target_include_directories(clock_format PRIVATE "${UI}")
add_test(NAME clock_format COMMAND clock_format)

add_library(watch_core STATIC "${CORE}/watch_core.c")
target_include_directories(watch_core PUBLIC "${CORE}")
add_executable(core_navigation core_navigation.c)
target_link_libraries(core_navigation PRIVATE watch_core)
add_test(NAME core_navigation COMMAND core_navigation)

file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL}/src/*.c")
add_library(lvgl_host STATIC ${LVGL_SOURCES})
target_include_directories(lvgl_host PUBLIC "${LVGL}")
target_compile_definitions(lvgl_host PUBLIC LV_CONF_PATH="${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" LV_LVGL_H_INCLUDE_SIMPLE)
if(WRISTFLOW_BUILD_SIMULATOR)
    target_compile_definitions(lvgl_host PUBLIC LV_USE_WINDOWS=1)
    target_link_libraries(lvgl_host PUBLIC gdi32 user32 imm32)
endif()
include("${UI}/file_list_gen.cmake")
add_library(wristflow_ui_host STATIC ${LV_EDITOR_PROJECT_SOURCES}
    "${UI}/clock_format.c" "${UI}/watchface_presenter.c" "${DEMO}/ui_demo.c"
    "${RUNTIME}/mount_screen.c" "${RUNTIME}/default_watchface.c" "${RUNTIME}/ui_shell.c")
target_include_directories(wristflow_ui_host PUBLIC "${UI}" "${DEMO}" "${RUNTIME}")
target_link_libraries(wristflow_ui_host PUBLIC lvgl_host watch_core)
add_executable(ui_preview ui_preview.c)
target_link_libraries(ui_preview PRIVATE wristflow_ui_host)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/renders")
add_test(NAME ui_preview COMMAND ui_preview "${CMAKE_CURRENT_BINARY_DIR}/renders")
add_executable(ui_navigation ui_navigation.c)
target_link_libraries(ui_navigation PRIVATE wristflow_ui_host)
add_test(NAME ui_navigation COMMAND ui_navigation "${CMAKE_CURRENT_BINARY_DIR}/renders")
add_executable(ui_shell_lifecycle ui_shell_lifecycle.c)
target_link_libraries(ui_shell_lifecycle PRIVATE wristflow_ui_host)
add_test(NAME ui_shell_lifecycle COMMAND ui_shell_lifecycle)
foreach(test_target clock_format core_navigation ui_preview ui_navigation ui_shell_lifecycle)
    if(MSVC)
        target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
endforeach()
if(WRISTFLOW_BUILD_SIMULATOR)
    add_executable(wristflow_simulator WIN32 "${REPO}/apps/simulator/main.c")
    target_link_libraries(wristflow_simulator PRIVATE wristflow_ui_host)
endif()
